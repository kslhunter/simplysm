#### 1. 공유 헬퍼 파일 생성

위치: `tests/angular/_vite-angular-plugin-test-setup.ts` [근거: Feature 1.1 D3 패턴]

```typescript
import { vi } from "vitest";
import path from "path";
import type { SdConfig, SdClientPackageConfig } from "../../src/sd-config.types";

export const FIXTURE_DIR = path.resolve(import.meta.dirname, "fixtures");
export const PKG_DIR = path.resolve(FIXTURE_DIR, "packages/basic-app");

export function createTestSdConfig(
  overrides?: Partial<SdClientPackageConfig>,
): SdConfig {
  return {
    packages: {
      "basic-app": { target: "client", server: 3000, ...overrides },
    },
  };
}

/** Vite lifecycle 시뮬레이션: config → configResolved → [configureServer] → buildStart */
export async function initPlugin(
  plugin: any,
  options?: { mode?: string; command?: string; sourcemap?: boolean; withServer?: boolean },
): Promise<void> {
  const mode = options?.mode ?? "development";
  const command = options?.command ?? "serve";
  const sourcemap = options?.sourcemap ?? false;
  await plugin.config?.({}, { mode, command });
  plugin.configResolved?.({ build: { sourcemap } });
  if (options?.withServer === true) {
    plugin.configureServer?.({
      middlewares: { use: vi.fn() },
      httpServer: { on: vi.fn() },
      config: { base: "/" },
      hot: { send: vi.fn() },
    });
  }
  await plugin.buildStart?.call({});
}
```

### 대안 검토

**각 파일에 헬퍼 인라인 유지**: 공유 파일 없이 각 파일에서 createTestSdConfig/initPlugin을 유지 → 7개 파일에서 동일 코드 ~30줄 반복. Feature 1.1에서 공유 헬퍼 패턴이 이미 확립되어 있으므로 일관성을 위해 공유 파일 생성이 적절. [근거: Feature 1.1 D3에서 _angular-compiler-test-setup.ts 생성]

**scss-hmr의 beforeAll을 beforeEach로 변경**: 테스트 격리 강화 → scss-hmr의 3개 테스트는 서로 다른 SCSS 파일(_variables.scss, _colors.scss, _unrelated.scss)을 사용하므로 상태 간섭 없음. Angular AOT 컴파일이 ~수초 소요되어 beforeEach는 불필요한 성능 저하. [근거: 코드 확인 — 테스트 간 파일 경로 중복 없음]
